59 // Implement Receiver method
60 virtual bool initializeReceiver()
61 {
62 bool ok = true;
63 // Resolve host:port (Asio getaddrinfo, or c-ares when QUICKFAST_HAVE_CARES).
64 asio::error_code error;
65 const std::vector<asio::ip::tcp::endpoint> endpoints =
66 resolveTcp(ioService_, hostName_, port_, error);
67 if(error || endpoints.empty())
68 {
69 ok = false;
70 std::stringstream msg;
71 msg << "Cannot resolve [" << hostName_ << ':' << port_ << "]: " << error;
72 (void) assembler_->reportCommunicationError(msg.str());
73 return ok;
74 }
75
76 // then iterate thru the collection until we find one that works.
77 bool connected = false;
78 for(std::vector<asio::ip::tcp::endpoint>::const_iterator iterator = endpoints.begin();
79 !connected && iterator != endpoints.end();
80 ++iterator)
81 {
82 socket_.connect(*iterator, error);
83 connected = !error;
84 }
85 if(!connected)
86 {
87 ok = false;
88 std::stringstream msg;
89 msg << "Cannot connect to [" << hostName_ << ':' << port_ << "]: " << error;
90 (void) assembler_->reportCommunicationError(msg.str());
91 }
92 else
93 {
94 if(assembler_->wantLog(Common::Logger::QF_LOG_INFO))
95 {
96 std::stringstream msg;
97 msg << "Connected to: " << hostName_ << ':' << port_;
98 assembler_->logMessage(Common::Logger::QF_LOG_INFO, msg.str());
99 }
100 }
101 return ok;
102 }
