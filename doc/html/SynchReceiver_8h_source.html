57 bool acceptFullBuffer(
58 LinkedBuffer * buffer,
59 size_t bytesReceived,
60 std::unique_lock<std::mutex> & lock
61 )
62 {
63 bool needService = false;
64 --readsInProgress_;
65 ++packetsReceived_;
66 if(bytesReceived > 0)
67 {
68 ++packetsQueued_;
69 largestPacket_ = std::max<size_t>(largestPacket_, bytesReceived);
70 buffer->setUsed(bytesReceived);
71 buffer->stampReceiveTime();
72 needService = queue_.push(buffer, lock);
73 }
74 else
75 {
76 // empty buffer? just use it again
77 ++emptyPackets_;
78 idleBufferPool_.push(buffer);
79 }
80 return needService;
81 }
82
84 size_t tryServiceQueue()
85 {
86 size_t count = 0;
87 bool service = false;
88 { // Scope for lock
89 std::unique_lock<std::mutex> lock(bufferMutex_);
90 service = queue_.startService(lock);
91 }
92 while(service && !stopping_)
93 {
94 service = serviceQueue();
95 ++count;
96 }
97 if(inputComplete())
98 {
99 // The queue is drained, so end of input can finally become a stop.
100 // Doing it here rather than where the read failed is the whole of
101 // the fix: a file with fewer packets than buffers is read to EOF
102 // inside start(), before the caller's event loop has run once.
103 stop();
104 }
105 return count;
106 }

126 virtual size_t poll()
127 {
128 size_t count = 0;

130 while(more && !stopping_)
131 {
132 size_t pass = tryServiceQueue();
133 count += pass;
134 more = pass != 0;
135 }
136 return count;
137 }

139 virtual size_t poll_one()
140 {

145 }
146 return count;
147 }

149 virtual void runThreads([[maybe_unused]] size_t threadCount = 0, bool useThisThread = true)
150 {
151 if(useThisThread)
152 {
153 // If we're using this thread, that's all that is needed.
154 // so ignore threadCount
155 run();
156 }
157 else
158 {
159 // we have to start a thread, but it doesn't make sense to have
160 // more than one thread servicing a synchronous data source,
161 // so only start the one.
162 thread_.reset(
163 new std::thread([this]{ this->run(); }));
164 }
165 }
166
167 virtual void joinThreads()
168 {
169 if(bool(thread_))
170 {
171 thread_->join();
172 thread_.reset();
173 }
174 }
175 virtual bool waitBuffer()
176 {
177 // if it's going to be there, it's there.
178 return true;
179 }
