131 std::unique_lock<std::mutex> lock(freeMutex_);
132 Communication::LinkedBuffer * buffer = freeBuffers_.pop();
133 while(buffer == 0 && !stopping_)
134 {
135 freeWait_.wait(lock, [this]{ return !freeBuffers_.isEmpty() || stopping_; });
136 buffer = freeBuffers_.pop();
137 }

139 }
