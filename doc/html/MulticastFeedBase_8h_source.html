274 void handleReceive(
275 const asio::error_code& error,
276 LinkedBuffer * buffer,
277 size_t bytesReceived)
278 {
279 assert(readInProgress_);
280 readInProgress_ = false;
281 size_t accepted = bytesReceived;
282 if(!error && bytesReceived > 0 && !acceptSender(senderEndpoint_))
283 {
284 // Reported as an empty read so the host recycles the buffer and
285 // starts the next read; rejectedPackets_ is what says why.
286 ++rejectedPackets_;
287 accepted = 0;
288 }
289 host_.feedReceived(error, buffer, accepted);
290 if(host_.feedStopping())
291 {
292 if(joined_)
293 {
294 leave();
295 joined_ = false;
296 }
297 socket_.close();
298 }
299 }
