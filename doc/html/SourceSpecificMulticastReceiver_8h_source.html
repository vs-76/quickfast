60 SourceSpecificMulticastFeed(
61 MulticastFeedHost & host,
62 AsioService & ioService,
63 const std::string & name,
64 const std::string & multicastGroupIP,
65 const SourceList & sourceIPs,
66 const std::string & listenInterfaceIP,
67 const std::string & bindIP,
68 unsigned short portNumber
69 )
70 : MulticastFeedBase(
71 host,
72 ioService,
73 name,
74 multicastGroupIP,
75 listenInterfaceIP,
76 bindIP,
77 portNumber)
78 {
79 if(sourceIPs.empty())
80 {
81 throw std::invalid_argument(
82 "Source specific multicast feed " + name + " needs at least one source address");
83 }
84 requireV4(multicastGroup(), "multicast group", name);
85 requireV4(listenInterface(), "listen interface", name);
86 requireV4(bindAddress(), "bind address", name);
87 for(size_t nSource = 0; nSource < sourceIPs.size(); ++nSource)
88 {
89 asio::ip::address source = asio::ip::make_address(sourceIPs[nSource]);
90 requireV4(source, "source address", name);
91 sources_.push_back(source.to_v4());
92 }
93 }

96 const std::vector<asio::ip::address_v4> & sources()const
97 {
98 return sources_;
99 }

108 static bool kernelSourceFilter()
109 {
110 return QUICKFAST_HAS_SOURCE_SPECIFIC_MULTICAST != 0;
111 }

114 // Implement MulticastFeedBase
115 virtual void join()
116 {
117#if QUICKFAST_HAS_SOURCE_SPECIFIC_MULTICAST
118 size_t joined = 0;
119 try
120 {
121 for(joined = 0; joined < sources_.size(); ++joined)
122 {
123 MulticastSource::join_source_group joinRequest(
124 multicastGroup().to_v4(),
125 sources_[joined],
126 listenInterface().to_v4());
127 socket().set_option(joinRequest);
128 }
129 }
130 catch(...)
131 {
132 dropSources(joined);
133 throw;
134 }
135#else
136 asio::ip::multicast::join_group joinRequest(
137 multicastGroup().to_v4(),
138 listenInterface().to_v4());
139 socket().set_option(joinRequest);
140#endif // QUICKFAST_HAS_SOURCE_SPECIFIC_MULTICAST
141 }
142
143 virtual void leave()
144 {
145#if QUICKFAST_HAS_SOURCE_SPECIFIC_MULTICAST
146 dropSources(sources_.size());
147#else
148 asio::ip::multicast::leave_group leaveRequest(
149 multicastGroup().to_v4(),
150 listenInterface().to_v4());
151 socket().set_option(leaveRequest);
152#endif // QUICKFAST_HAS_SOURCE_SPECIFIC_MULTICAST
153 }
154
155 virtual bool acceptSender(const asio::ip::udp::endpoint & sender) const
156 {
157 const asio::ip::address & address = sender.address();
158 if(!address.is_v4())
159 {
160 return false;
161 }
162 const asio::ip::address_v4 v4 = address.to_v4();
163 for(size_t nSource = 0; nSource < sources_.size(); ++nSource)
164 {
165 if(sources_[nSource] == v4)
166 {
167 return true;
168 }
169 }
170 return false;
171 }
172
173 private:
174 static void requireV4(
175 const asio::ip::address & address,
176 const char * role,
177 const std::string & name)
178 {
179 if(!address.is_v4())
180 {
181 throw std::invalid_argument(
182 "Source specific multicast feed " + name + ": " + role + " "
183 + address.to_string() + " is not IPv4");
184 }
185 }
186
187#if QUICKFAST_HAS_SOURCE_SPECIFIC_MULTICAST
191 void dropSources(size_t count)
192 {
193 for(size_t nSource = 0; nSource < count; ++nSource)
194 {
195 MulticastSource::leave_source_group leaveRequest(
196 multicastGroup().to_v4(),
197 sources_[nSource],
198 listenInterface().to_v4());
199 asio::error_code ec;
200 socket().set_option(leaveRequest, ec);
201 }
202 }
203#endif // QUICKFAST_HAS_SOURCE_SPECIFIC_MULTICAST
204
205 private:
206 std::vector<asio::ip::address_v4> sources_;
207 };

346 private:
347 void warnAboutGroupsOutsideTheSSMRange()
348 {
349 if(!assembler_->wantLog(Common::Logger::QF_LOG_WARNING))
350 {
351 return;
352 }
353 for(size_t nFeed = 0; nFeed < feedCount(); ++nFeed)
354 {
355 MulticastFeedBase & candidate = feed(nFeed);
356 const asio::ip::address group = candidate.multicastGroup();
357 if(group.is_v4() && !MulticastSource::isSourceSpecificRange(group.to_v4()))
358 {
359 std::stringstream msg;
360 msg << "Feed " << candidate.name() << ": group " << group.to_string()
361 << " is outside the source specific range 232.0.0.0/8."
362 << " Source filtering depends on the network honoring IGMPv3 there.";
363 assembler_->logMessage(Common::Logger::QF_LOG_WARNING, msg.str());
364 }
365 }
366 }
