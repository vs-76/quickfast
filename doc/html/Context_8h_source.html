189 template<typename VALUE_TYPE>
190 DictionaryStatus getDictionaryValue(size_t index, VALUE_TYPE & value)
191 {
192 if(index > indexedDictionarySize_)
193 {
194 throw TemplateDefinitionError("Illegal dictionary index.");
195 }
196 Value & entry = indexedDictionary_[index];
197 if(!entry.isDefined())
198 {
199 return UNDEFINED_VALUE;
200 }
201 if(entry.isNull())
202 {
203 return NULL_VALUE;
204 }
205 (void)entry.getValue(value);
206 return OK_VALUE;
207 }
