18// Note: boost's microsecond_clock was slow enough to skew the results
19// hence the win32 GetTickCount stuff.
20#ifdef _WIN32
21# ifndef _WINBASE_ // if we didn't #include <windows.h>
22// this is the ultimate expansion of the declaration for GetTickCount
23// after the macros have been resolved on a 16 bit windows build
24// With luck, it won't conflict with the windows declaration in case
25// the included occur in the wrong order
26typedef unsigned long DWORD;
27__declspec(dllimport) DWORD GetTickCount(void);
28# endif // _WINBASE_
29# define PROFILER_GET_TIME ::GetTickCount()
30# define PROFILER_TIME_TYPE DWORD
31# define PROFILER_DIFF_MSEC(a, b) ((a) - (b))
32#else // _WIN32
33# define PROFILER_GET_TIME std::chrono::steady_clock::now()
34# define PROFILER_TIME_TYPE std::chrono::steady_clock::time_point
35# define PROFILER_DIFF_MSEC(a,b) (std::chrono::duration_cast<std::chrono::milliseconds>((a) - (b)).count())
36#endif // _WIN32

89 ProfileInstance(ProfileAccumulator & accumulator)
90 : accumulator_(accumulator)
91 , start_(PROFILER_GET_TIME)
92 , running_(true)
93 {
94 accumulator_.entries_ += 1;
95 }
96
98 ~ProfileInstance()
99 {
100 stop();
101 accumulator_.exits_ += 1;
102 }
103
106 bool pause()
107 {
108 bool result = running_;
109 stop();
110 accumulator_.pauses_ += 1;
111 return result;
112 }
113
116 void resume(bool pauseState)
117 {
118 accumulator_.resumes_ += 1;
119 if(!running_ && pauseState)
120 {
121 start_ = PROFILER_GET_TIME;
122 running_ = true;
123 }
124 }
125
126 private:
127 void stop()
128 {
129 if(running_)
130 {
131 PROFILER_TIME_TYPE now = PROFILER_GET_TIME;
132 double lapse = double(PROFILER_DIFF_MSEC(now, start_));
133 accumulator_.sum_ += lapse;
134 accumulator_.sumOfSquares_ += lapse * lapse;
135 assert(accumulator_.entries_ > accumulator_.exits_);
136 if(accumulator_.entries_ != accumulator_.exits_ + 1)
137 {
138 accumulator_.recursions_ += 1;
139 accumulator_.recursiveSum_ += lapse;
140 accumulator_.recursiveSumOfSquares_ += lapse * lapse;
141 }
142 running_ = false;
143 }
144 }
145 private:
146 ProfileInstance & operator =(const ProfileInstance &);
147 ProfileInstance();
148 ProfileInstance(const ProfileInstance &);
149
150 private:
151 ProfileAccumulator & accumulator_;
152 PROFILER_TIME_TYPE start_;
153 bool running_;
154 };
