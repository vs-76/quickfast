99 bool empty = false;
100 {
101 std::unique_lock<std::mutex> lock(bufferMutex_);
102 //if(!idleBuffers_.isEmpty())

105 // msg << "AR:{"<< (void *) this << "} making idle buffers available" << std::endl;
106 // std::cout << msg.str() << std::flush;
107 //}
108 idleBufferPool_.push(idleBuffers_);
109 // be sure we have a read request in progress
110 startReceive(lock);
111 // promote any ancoming messages to outgoing
112 empty = !queue_.refresh(lock, ioService_.runningThreadCount() != 1 && !stopping_);

120 if(empty && ioService_.runningThreadCount() == 1 && !stopping_)

152 void handleReceive(
153 const asio::error_code& error,
154 LinkedBuffer * buffer,
155 size_t bytesReceived)
156 {
157 // should this thread service the queue?
158 bool service = false;
159 { // Scope for lock
160 std::unique_lock<std::mutex> lock(bufferMutex_);
161 --readsInProgress_;
162 ++packetsReceived_;
163 if (!error)
164 {
165 // it's possible to receive empty packets.
166 if(bytesReceived <= 0)
167 {
168 // empty buffer? just use it again
169 ++emptyPackets_;
170 idleBufferPool_.push(buffer);
171 }
172 else if(paused_)
173 {
174 // We're paused. Ignore incoming packets
175 ++pausedPackets_;
176 idleBufferPool_.push(buffer);
177 }
178 else
179 {
180 ++packetsQueued_;
181 bytesReceived_ += bytesReceived;
182 largestPacket_ = std::max<size_t>(largestPacket_, bytesReceived);
183 buffer->setUsed(bytesReceived);
184 buffer->stampReceiveTime();
185 if(queue_.push(buffer, lock))
186 {
187 // A true return from push means that no one is servicing the queue
188 // Volunteer to service it. If it returns true, then the offer was
189 // accepted. We'll service the queue after releasing the lock.
190 service = queue_.startService(lock);
191 }
192 }
193 }
194 else
195 {
196 // after an error, recover the buffer
197 idleBufferPool_.push(buffer);
198 // ignore errors during state transitions
199 if(!paused_ && !stopping_)
200 {
201 ++errorPackets_;
202 // and let the consumer decide what to do
203 if(!assembler_->reportCommunicationError(error.message()))
204 {
205 stop();
206 }
207 }
208 }
209 // if possible fill another buffer while we process this one
210 startReceive(lock);
211 // end of scope for lock
212 }
213
214 while(service)
215 {
216 service = serviceQueue();
217 }
218 }
219
220 protected:
222 AsioService ioService_;
223 };
