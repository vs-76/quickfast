298 void setMulticastName(const std::string & name)
299 {
300 if(multicastFeeds_.size() == 1 && multicastFeeds_[0].name_ == DEFAULT_MULTICAST_NAME)
301 {
302 multicastFeeds_[0].name_ = name;

900 int parseSingleArg(int argc, char * argv[])
901 {
902 int consumed = 0;
903 std::string opt(argv[0]);
904 if(opt == "-t" && argc > 1)
905 {
906 setTemplateFileName(argv[1]);
907 consumed = 2;
908 }
909 else if(opt == "-limit" && argc > 1)
910 {
911 setHead(QuickFAST::lexical_cast<size_t>(argv[1]));
912 consumed = 2;
913 }
914 else if(opt == "-reset")
915 {
916 setReset(true);
917 consumed = 1;
918 }
919 else if(opt == "-strict")
920 {
921 setStrict(false);
922 consumed = 1;
923 }
924 else if(opt == "-vo" && argc > 1)
925 {
926 setVerboseFileName(argv[1]);
927 consumed = 2;
928 }
929 else if(opt == "-e" && argc > 1)
930 {
931 setEchoFileName(argv[1]);
932 consumed = 2;
933 }
934 else if(opt == "-ehex")
935 {
936 setEchoType(Codecs::DataSource::HEX);
937 consumed = 1;
938 }
939 else if(opt == "-eraw")
940 {
941 setEchoType(Codecs::DataSource::RAW);
942 consumed = 1;
943 }
944 else if(opt == "-enone")
945 {
946 setEchoType(Codecs::DataSource::NONE);
947 consumed = 1;
948 }
949 else if(opt == "-em")
950 {
951 setEchoMessage(true);
952 consumed = 1;
953 }
954 else if(opt == "-em-")
955 {
956 setEchoMessage(false);
957 consumed = 1;
958 }
959 else if(opt == "-ef")
960 {
961 setEchoField(true);
962 consumed = 1;
963 }
964 else if(opt == "-ef-")
965 {
966 setEchoField(false);
967 consumed = 1;
968 }
969 else if(opt == "-file" && argc > 1)
970 {
971 setReceiverType(RAWFILE_RECEIVER);
972 setFastFileName(argv[1]);
973 consumed = 2;
974 }
975 else if(opt == "-afile" && argc > 1)
976 {
977 setReceiverType(ASYNCHRONOUS_FILE_RECEIVER);
978 setFastFileName(argv[1]);
979 setAsynchReads(true);
980 consumed = 2;
981 }
982 else if(opt == "-bfile" && argc > 1)
983 {
984 setReceiverType(BUFFERED_RAWFILE_RECEIVER);
985 setFastFileName(argv[1]);
986 consumed = 2;
987 }
988 else if(opt == "-pcap" && argc > 1)
989 {
990 setReceiverType(PCAPFILE_RECEIVER);
991 setPcapFileName(argv[1]);
992 consumed = 2;
993 }
994 else if(opt == "-mname" && argc > 1)
995 {
996 setMulticastName(argv[1]);
997 consumed = 2;
998 }
999 else if(opt == "-multicast" && argc > 1)
1000 {
1001 setReceiverType(MULTICAST_RECEIVER);
1002 std::string address = argv[1];
1003 std::string::size_type colon = address.find(':');
1004 setMulticastGroupIP(address.substr(0, colon));
1005 if(colon != std::string::npos)
1006 {
1007 setPortNumber(QuickFAST::lexical_cast<unsigned short>(
1008 address.substr(colon+1)));
1009 }
1010 consumed = 2;
1011 }
1012 else if(opt == "-mlisten" && argc > 1)
1013 {
1014 setListenInterfaceIP(argv[1]);
1015 consumed = 2;
1016 }
