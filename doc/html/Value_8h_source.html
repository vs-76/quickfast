258 bool operator == (const Value & rhs) const
259 {

573 void valueToStringBuffer()const
574 {
575 if(cachedString_ || (class_ == STRING))
576 {
577 return;
578 }
579 std::stringstream buffer;
580 if(class_ == SIGNEDINTEGER)
581 {
582 buffer << signedInteger_;
583 }
584 else if(class_ == UNSIGNEDINTEGER)
585 {
586 buffer << unsignedInteger_;
587 }
588 else if(class_ == DECIMAL)
589 {
590 Decimal d(signedInteger_, exponent_);
591
592 buffer << static_cast<double>(d);
593 }
594 else if(class_ == EMPTY)
595 {
596 buffer << "[null]";
597 }
598 ensureString() = buffer.str();
599 }
