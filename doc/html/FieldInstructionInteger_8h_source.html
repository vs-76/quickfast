211 void applyDelta(
212 Codecs::Decoder & decoder, INTEGER_TYPE & value, int64 delta) const
213 {
214 const uint64 sum = uint64(value) + uint64(delta);
215 const INTEGER_TYPE narrowed = INTEGER_TYPE(sum);
216 if(uint64(narrowed) != sum)
217 {
218 decoder.reportError("[ERR D2]",
219 "Applying delta leaves the range of the field.", identity_);
220 return;
221 }
222 value = narrowed;
223 }

259 template<typename INTEGER_TYPE, ValueType::Type VALUE_TYPE, bool SIGNED>
260 void
261 FieldInstructionInteger<INTEGER_TYPE, VALUE_TYPE, SIGNED>::
262 interpretValue(const std::string & value)
263 {
264 if(value.empty())
265 {
266 typedValue_ = INTEGER_TYPE(0);
267 typedValueIsDefined_ = false;
268 }
269 else
270 {
271 if(SIGNED)
272 {
273 int64 v = QuickFAST::lexical_cast<int64>(value);
274 typedValue_ = INTEGER_TYPE(v);
275 if(v != int64(typedValue_))
276 {
277 throw QuickFAST::TemplateDefinitionError("[ERR D1] Initial value overflow");
278 }
279 typedValueIsDefined_ = true;
280 }
281 else
282 {
283 uint64 v = QuickFAST::lexical_cast<uint64>(value);
284 typedValue_ = INTEGER_TYPE(v);
285 if(v != uint64(typedValue_))
286 {
287 throw QuickFAST::TemplateDefinitionError("[ERR D1] Initial value overflow");
288 }
289 typedValueIsDefined_ = true;
290 }
291 }

294 template<typename INTEGER_TYPE, ValueType::Type VALUE_TYPE, bool SIGNED>
295 bool
296 FieldInstructionInteger<INTEGER_TYPE, VALUE_TYPE, SIGNED>::
297 decodeNullableValue(
298 Codecs::DataSource & source,
299 Codecs::Decoder & decoder,
300 INTEGER_TYPE & value) const
301 {
302 if(isMandatory())
303 {
304 if constexpr(SIGNED)
305 {
306 decodeSignedInteger(source, decoder, value, identity_.name(), false, ignoreOverflow_);
307 }
308 else
309 {
310 decodeUnsignedInteger(source, decoder, value, identity_.name(), ignoreOverflow_);
311 }
312 return true;
313 }
314
315 if constexpr(SIGNED)
316 {
317 int64 wide = 0;
318 decodeSignedInteger(source, decoder, wide, identity_.name(), true, ignoreOverflow_);
319 if(checkNullInteger(wide))
320 {
321 return false;
322 }
323 if(!ignoreOverflow_ &&
324 (wide < int64(std::numeric_limits<INTEGER_TYPE>::min()) ||
325 wide > int64(std::numeric_limits<INTEGER_TYPE>::max())))
326 {
327 decoder.reportError("[ERR D2]", "Signed Integer Field overflow.", identity_);
328 }
329 value = static_cast<INTEGER_TYPE>(wide);
330 }
331 else
332 {
333 uint64 wide = 0;
334 decodeUnsignedInteger(source, decoder, wide, identity_.name(), ignoreOverflow_);
335 if(checkNullInteger(wide))
336 {
337 return false;
338 }
339 if(!ignoreOverflow_ && wide > uint64(std::numeric_limits<INTEGER_TYPE>::max()))
340 {
341 decoder.reportError("[ERR D2]", "Unsigned Integer Field overflow.", identity_);
342 }
343 value = static_cast<INTEGER_TYPE>(wide);
344 }
345 return true;
346 }

406 template<typename INTEGER_TYPE, ValueType::Type VALUE_TYPE, bool SIGNED>
407 void
408 FieldInstructionInteger<INTEGER_TYPE, VALUE_TYPE, SIGNED>::
409 decodeCopy(
410 Codecs::DataSource & source,
411 bool pmapValue,
412 Codecs::Decoder & decoder,
413 Messages::ValueMessageBuilder & builder) const
414 {
415 PROFILE_POINT("int::decodeCopy");
416 if(pmapValue)
417 {
418 INTEGER_TYPE value = typedValue_;
419 // present in stream
420 if(decodeNullableValue(source, decoder, value))
421 {
422 builder.addValue(
423 identity_,
424 VALUE_TYPE,
425 value);
426 fieldOp_->setDictionaryValue(decoder, value);
427 }
428 else
429 {
430 fieldOp_->setDictionaryValueNull(decoder);
431 }
432 }
433 else // pmap says not present, use copy
434 {
435 INTEGER_TYPE previousValue = 0;
436 Context::DictionaryStatus previousStatus = fieldOp_->getDictionaryValue(decoder, previousValue);
437 if(previousStatus == Context::OK_VALUE)
438 {
439 builder.addValue(
440 identity_,
441 VALUE_TYPE,
442 previousValue);
443 }
444 else if(previousStatus == Context::UNDEFINED_VALUE)
445 {
446 // value not found in dictionary
447 // not a problem.. use initial value if it's available
448 if(fieldOp_->hasValue())
449 {
450 builder.addValue(
451 identity_,
452 VALUE_TYPE,
453 typedValue_);
454 fieldOp_->setDictionaryValue(decoder, typedValue_);
455 }
456 else
457 {
458 if(isMandatory())
459 {
460 decoder.reportError(
461 "[ERR D5]",
462 "Copy operator missing mandatory integer field/no initial value",
463 identity_);
464 builder.addValue(
465 identity_,
466 VALUE_TYPE,
467 INTEGER_TYPE(0));
468 fieldOp_->setDictionaryValue(decoder, INTEGER_TYPE(0));
469 }
470 }
471 }
472 else // NULL value
473 {
474 if(isMandatory())
475 {
476 // This is unlikely, but if two fields share a dictionary entry and one
477 // is mandatory while the other is optional....
478 decoder.reportError(
479 "[ERR D5]",
480 "Copy operator mandatory integer field, but previous value was NULL",
481 identity_);
482 builder.addValue(
483 identity_,
484 VALUE_TYPE,
485 INTEGER_TYPE(0));
486 fieldOp_->setDictionaryValue(decoder, INTEGER_TYPE(0));
487 }
488 }
489 }
490 }

493 void
494 FieldInstructionInteger<INTEGER_TYPE, VALUE_TYPE, SIGNED>::
495 decodeDefault(
496 Codecs::DataSource & source,
497 Codecs::PresenceMap & pmap,
498 Codecs::Decoder & decoder,
499 Messages::ValueMessageBuilder & builder) const
500 {
501 PROFILE_POINT("int::decodeDefault");
502 if(pmap.checkNextField())
503 {
504 PROFILE_POINT("int::decodeDefault:present");
505 INTEGER_TYPE value = 0;
506 if(decodeNullableValue(source, decoder, value))
507 {
508 PROFILE_POINT("int::decodeDefault:,addexplicit");
509 builder.addValue(
510 identity_,
511 VALUE_TYPE,
512 value);
513 }
514 }
515 else // field not in stream
516 {
517 PROFILE_POINT("int::decodeDefault:absent");
518 if(fieldOp_->hasValue())
519 {
520 PROFILE_POINT("int::decodeDefault:adddefault");
521 builder.addValue(
522 identity_,
523 VALUE_TYPE,
524 typedValue_);
525 }
526 else if(isMandatory())
527 {
528 decoder.reportError("[ERR D5]", "Mandatory default operator with no value.", identity_);
529 }
530 }

533 template<typename INTEGER_TYPE, ValueType::Type VALUE_TYPE, bool SIGNED>
534 void
535 FieldInstructionInteger<INTEGER_TYPE, VALUE_TYPE, SIGNED>::
536 decodeDelta(
537 Codecs::DataSource & source,
538 Codecs::PresenceMap & /*pmap*/,
539 Codecs::Decoder & decoder,
540 Messages::ValueMessageBuilder & builder) const
541 {
542 PROFILE_POINT("int::decodeDelta");
543 int64 delta;
544 decodeSignedInteger(source, decoder, delta, identity_.name(), true);
545 if(!isMandatory())
546 {
547 if(checkNullInteger(delta))
548 {
549 return; // nothing in Message; no change to saved value
550 }
551 }
552 INTEGER_TYPE value = typedValue_;
553 Context::DictionaryStatus previousStatus = fieldOp_->getDictionaryValue(decoder, value);
554 if(previousStatus == Context::UNDEFINED_VALUE)
555 {
556 if(fieldOp_->hasValue()) // initial value in field op?
557 {
558 value = typedValue_;
559 }
560 }
561 // Apply delta
562 applyDelta(decoder, value, delta);
563 // Save the results
564 builder.addValue(
565 identity_,
566 VALUE_TYPE,
567 value);
568 fieldOp_->setDictionaryValue(decoder, value);
569 }

584 template<typename INTEGER_TYPE, ValueType::Type VALUE_TYPE, bool SIGNED>
585 void
586 FieldInstructionInteger<INTEGER_TYPE, VALUE_TYPE, SIGNED>::
587 decodeIncrement(
588 Codecs::DataSource & source,
589 bool pmapValue,
590 Codecs::Decoder & decoder,
591 Messages::ValueMessageBuilder & builder) const
592 {
593 PROFILE_POINT("int::decodeIncrement");
594 if(pmapValue)
595 {
596 //PROFILE_POINT("int::decodeIncrement::present");
597 INTEGER_TYPE value = 0;
598 if(decodeNullableValue(source, decoder, value))
599 {
600 builder.addValue(
601 identity_,
602 VALUE_TYPE,
603 value);
604 fieldOp_->setDictionaryValue(decoder, value);
605 }
606 else
607 {
608 fieldOp_->setDictionaryValueNull(decoder);
609 }
610 }
611 else
612 {
613 //PROFILE_POINT("int::decodeIncrement::absent");
614 INTEGER_TYPE value = typedValue_;
615 Context::DictionaryStatus previousStatus = fieldOp_->getDictionaryValue(decoder, value);
616 if(previousStatus == Context::OK_VALUE)
617 {
618 // The increment operator exists for monotonically rising fields, so
619 // a sequence number reaching the type maximum is where a
620 // long-running feed ends up rather than an attack. It used to
621 // continue at zero, or invoke undefined behaviour for the signed
622 // instantiations.
623 if(value == std::numeric_limits<INTEGER_TYPE>::max())
624 {
625 decoder.reportError("[ERR D2]",
626 "Increment leaves the range of the field.", identity_);
627 }
628 value = INTEGER_TYPE(uint64(value) + 1);
629 }
630 else if(previousStatus == Context::UNDEFINED_VALUE)
631 {
632 if(fieldOp_->hasValue())
633 {
634 value = typedValue_;
635 }
636 else
637 {
638 if(isMandatory())
639 {
640 decoder.reportError("[ERR D5]", "Missing initial value for mandatory integer with increment operator", identity_);
641 value = 0;
642 }
643 else
644 {
645 // missing value for optional field. We're done
646 return;
647 }
648 }
649 }
650 else // prevousStatus = NULL_VALUE
651 {
652 if(isMandatory())
653 {
654 decoder.reportError("[ERR D5]", "Null value for mandatory integer with increment operator", identity_);
655 value = 0;
656 }
657 else
658 {
659 // missing value for optional field. We're done
660 return;
661 }
662 }
663 builder.addValue(
664 identity_,
665 VALUE_TYPE,
666 value);
667 fieldOp_->setDictionaryValue(decoder, value);
668 }
669 }

671 template<typename INTEGER_TYPE, ValueType::Type VALUE_TYPE, bool SIGNED>
672 void
673 FieldInstructionInteger<INTEGER_TYPE, VALUE_TYPE, SIGNED>::
674 encodeNullableValue(
675 Codecs::DataDestination & destination,
676 Codecs::Encoder & encoder,
677 INTEGER_TYPE value) const
678 {
679 if(SIGNED)
680 {
681 int64 wire = static_cast<int64>(value);
682 if(!isMandatory() && wire >= 0)
683 {
684 if(wire == std::numeric_limits<int64>::max())
685 {
686 encoder.reportError("[ERR D2]",
687 "Value too large to encode in an optional signed integer field.",
688 identity_);
689 // Only reached when the application asked to continue past errors.
690 // An explicit null keeps the stream well formed and says "no value"
691 // rather than quietly substituting the value's own negation.
692 destination.putByte(nullInteger);
693 return;
694 }
695 ++wire;
696 }
697 encodeSignedInteger(destination, encoder.getWorkingBuffer(), wire);
698 }
699 else
700 {
701 uint64 wire = static_cast<uint64>(value);
702 if(!isMandatory())
703 {
704 if(wire == std::numeric_limits<uint64>::max())
705 {
706 encoder.reportError("[ERR D2]",
707 "Value too large to encode in an optional unsigned integer field.",
708 identity_);
709 destination.putByte(nullInteger);
710 return;
711 }
712 ++wire;
713 }
714 encodeUnsignedInteger(destination, encoder.getWorkingBuffer(), wire);
715 }
716 }

718 template<typename INTEGER_TYPE, ValueType::Type VALUE_TYPE, bool SIGNED>
719 void
720 FieldInstructionInteger<INTEGER_TYPE, VALUE_TYPE, SIGNED>::
721 encodeNop(
722 Codecs::DataDestination & destination,
723 Codecs::PresenceMap & /*pmap*/,
724 Codecs::Encoder & encoder,
725 const Messages::MessageAccessor & accessor) const
726 {
727 // get the value from the application data
728 INTEGER_TYPE value = 0;
729 bool present = 0;
730 if(SIGNED)
731 {
732 int64 value64;
733 present = accessor.getSignedInteger(identity_, VALUE_TYPE, value64);
734 value = static_cast<INTEGER_TYPE>(value64);
735 }
736 else
737 {
738 uint64 value64;
739 present = accessor.getUnsignedInteger(identity_, VALUE_TYPE, value64);
740 value = static_cast<INTEGER_TYPE>(value64);
741 }
742
743 if(present)
744 {
745 encodeNullableValue(destination, encoder, value);
746 }
747 else // not defined in fieldset
748 {
749 if(isMandatory())
750 {
751 encoder.reportError("[ERR U01]", "Missing mandatory field.", identity_);
752 if(SIGNED)
753 {
754 encodeSignedInteger(destination, encoder.getWorkingBuffer(), 0);
755 }
756 else
757 {
758 encodeUnsignedInteger(destination, encoder.getWorkingBuffer(), 0);
759 }
760 }
761 else
762 {
763 destination.putByte(nullInteger);
764 }
765 }

782 // get the value from the application data
783 INTEGER_TYPE value = 0;
784 bool present = 0;

788 present = accessor.getSignedInteger(identity_, VALUE_TYPE, value64);

795 value = static_cast<INTEGER_TYPE>(value64);
796 }
797
798 if(present)
799 {
800 if(value != typedValue_)
801 {
802 encoder.reportError("[ERR U02]", "Constant value does not match application data.", identity_);
803 }
804 pmap.setNextField(true);
805 }

808 pmap.setNextField(false);

827 int64 value64;
828 present = accessor.getSignedInteger(identity_, VALUE_TYPE, value64);

863 pmap.setNextField(false);
864 }

868 template<typename INTEGER_TYPE, ValueType::Type VALUE_TYPE, bool SIGNED>
869 void
870 FieldInstructionInteger<INTEGER_TYPE, VALUE_TYPE, SIGNED>::
871 encodeCopy(
872 Codecs::DataDestination & destination,
873 Codecs::PresenceMap & pmap,
874 Codecs::Encoder & encoder,
875 const Messages::MessageAccessor & accessor) const
876 {
877 // get previous value from dictionary
878 INTEGER_TYPE previousValue = 0;
879 Context::DictionaryStatus previousStatus = fieldOp_->getDictionaryValue(encoder, previousValue);
880 if(previousStatus == Context::UNDEFINED_VALUE)
881 {
882 if(fieldOp_->hasValue())
883 {
884 previousValue = typedValue_;
885 fieldOp_->setDictionaryValue(encoder, typedValue_);
886 // pretend we got the data from the dictionary
887 previousStatus = Context::OK_VALUE;
888 }
889 else
890 {
891 // pretend we got a null from the dictionary
892 fieldOp_->setDictionaryValueNull(encoder);
893 previousStatus = Context::NULL_VALUE;
894 }
895 }
896
897 // get the value from the application data
898 INTEGER_TYPE value = 0;
899 bool present = 0;
900 if(SIGNED)
901 {
902 int64 value64;
903 present = accessor.getSignedInteger(identity_, VALUE_TYPE, value64);
904 value = static_cast<INTEGER_TYPE>(value64);
905 }
906 else
907 {
908 uint64 value64;
909 present = accessor.getUnsignedInteger(identity_, VALUE_TYPE, value64);
910 value = static_cast<INTEGER_TYPE>(value64);
911 }
912
913 if(present)
914 {
915 if(previousStatus == Context::OK_VALUE && previousValue == value)
916 {
917 pmap.setNextField(false); // not in stream, use copy
918 }
919 else
920 {
921 pmap.setNextField(true);// value in stream
922 encodeNullableValue(destination, encoder, value);
923 fieldOp_->setDictionaryValue(encoder, value);
924 }
925 }

927 {

929 {
930 encoder.reportError("[ERR U01]", "Missing mandatory integer field.", identity_);

932 // Let the copy happen

953 template<typename INTEGER_TYPE, ValueType::Type VALUE_TYPE, bool SIGNED>
954 void
955 FieldInstructionInteger<INTEGER_TYPE, VALUE_TYPE, SIGNED>::
956 encodeDelta(
957 Codecs::DataDestination & destination,
958 Codecs::PresenceMap & /*pmap*/,
959 Codecs::Encoder & encoder,
960 const Messages::MessageAccessor & accessor) const
961 {
962 // get previous value from dictionary
963 INTEGER_TYPE previousValue = 0;
964 Context::DictionaryStatus previousStatus = fieldOp_->getDictionaryValue(encoder, previousValue);
965 if(previousStatus != Context::OK_VALUE && fieldOp_->hasValue())
966 {
967 previousValue = typedValue_;
968 fieldOp_->setDictionaryValue(encoder, typedValue_);
969 }
970
971 // get the value from the application data
972 INTEGER_TYPE value = 0;
973 bool present = 0;
974 if(SIGNED)
975 {
976 int64 value64;
977 present = accessor.getSignedInteger(identity_, VALUE_TYPE, value64);
978 value = static_cast<INTEGER_TYPE>(value64);
979 }
980 else
981 {
982 uint64 value64;
983 present = accessor.getUnsignedInteger(identity_, VALUE_TYPE, value64);
984 value = static_cast<INTEGER_TYPE>(value64);
985 }
986
987 if(present)
988 {
989 // int64(value) - int64(previousValue) converted a uint64 above
990 // INT64_MAX to a negative int64 before subtracting, and could overflow
991 // besides. The difference taken modulo 2^64 is the exact inverse of
992 // what applyDelta does on the far side, at every width and for both
993 // signednesses, and is well defined.
994 const uint64 difference = uint64(value) - uint64(previousValue);
995 int64 deltaValue = int64(difference);
996
997 if(!isMandatory() && deltaValue >= 0)
998 {
999 // FAST reserves zero for null, so a non-negative delta goes out one
1000 // larger. The addition used to be unguarded, and a delta already at
1001 // INT64_MAX came out as INT64_MIN -- a large positive jump arriving
1002 // as a large negative one.
1003 if(deltaValue == std::numeric_limits<int64>::max())
1004 {
1005 encoder.reportError("[ERR D2]",
1006 "Delta between consecutive values is too large for an optional"
1007 " field to encode.",
1008 identity_);
1009 return;
1010 }
1011 deltaValue += 1;
1012 }
1013 encodeSignedInteger(destination, encoder.getWorkingBuffer(), deltaValue);
1014 if(previousStatus != Context::OK_VALUE || value != previousValue)
1015 {
1016 fieldOp_->setDictionaryValue(encoder, value);
1017 }
1018
1019 }
1020 else // not defined in fieldset
1021 {
1022 if(isMandatory())
1023 {
1024 encoder.reportError("[ERR U01]", "Missing mandatory field.", identity_);
1025 encodeSignedInteger(destination, encoder.getWorkingBuffer(), 0);
1026 }
1027 else
1028 {
1029 destination.putByte(nullInteger);
1030 }
1031 }
1032 }
