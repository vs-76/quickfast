86 virtual const StringBuffer & displayString() const
87 {
88 if(valid_ && !isString())
89 {
90 // One lock for all fields: formatting is documented low performance,
91 // and a per-field mutex would grow an object the decoder creates
92 // millions of.
93 std::lock_guard<std::mutex> guard(displayStringMutex());
94 if(!stringCached_)
95 {
96 valueToStringBuffer();
97 // Keyed on a flag rather than on string_.empty(), so a value that
98 // formats to nothing is not re-rendered on every call.
99 stringCached_ = true;
100 }
101 }
102 return string_;
103 }
