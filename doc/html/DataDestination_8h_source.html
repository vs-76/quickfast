69 BufferHandle startBuffer()
70 {
71 if(used_ == buffers_.size())
72 {
73 WorkingBuffer empty;
74 buffers_.push_back(empty);
75 }
76 assert(used_ < buffers_.size());
77 active_ = used_++;
78 buffers_[active_].clear(false);
79 return active_;
80 }

84 void putByte(uchar byte)
85 {
86 if(active_ == NotABuffer)
87 {
88 (void)startBuffer();
89 }
90 buffers_[active_].push(byte);
91 if(verboseOut_)
92 {
93 (*verboseOut_)
94 << '[' << active_ << ':' << buffers_[active_].size() << ']'
95 << std::hex << std::setw(2) << std::setfill('0')
96 << static_cast<unsigned short>(byte)
97 << std::setfill(' ') << std::dec << ' ';
98 if(++verboseMod_ % 16 == 0) (*verboseOut_) << std::endl;
99 }
100 }

115 void selectBuffer(BufferHandle handle)
116 {
117 // BufferHandle is a bare size_t with no generation counter, so a
118 // handle obtained before a clear() is indistinguishable from a live
119 // one. Accepting it steered putByte into a slot outside the current
120 // message, where toString used to find it and send it. Checking
121 // against used_ rather than the pool size is what makes "the current
122 // message" mean something.
123 rangeCheck(handle, "selectBuffer");
124 active_ = handle;
125 }
126
128 void clear()
129 {
130 for(size_t handle = 0; handle < buffers_.size(); ++handle)
131 {
132 buffers_[handle].clear(false);
133 }
134 used_ = 0;
135 active_ = NotABuffer;
136 }

181 void toString(std::string & result)const
182 {
183 // used_, not buffers_.size(). The pool keeps the buffers from the
184 // longest cycle so far, and walking all of them reported slots that
185 // belong to no message -- disagreeing with size() and with the
186 // iterator pair, which have always counted what is in use.
187 size_t size = 0;
188 for(size_t pos = 0; pos < used_; ++pos)
189 {
190 size += buffers_[pos].size();
191 }
192 result.clear();
193 result.reserve(size);
194 for(size_t pos = 0; pos < used_; ++pos)
195 {
196 result.append(reinterpret_cast<const char *>(buffers_[pos].begin()),
197 buffers_[pos].size());
198 }
199 }
200
204 void toWorkingBuffer(WorkingBuffer & result) const
205 {
206 size_t size = 0;
207 for(size_t pos = 0; pos < used_; ++pos)
208 {
209 size += buffers_[pos].size();
210 }
211 result.clear(false, size);
212 for(size_t pos = 0; pos < used_; ++pos)
213 {
214 result.append(buffers_[pos]);
215 }
216 }
