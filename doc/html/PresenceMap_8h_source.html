135 void
136 PresenceMap::setNextField(bool present)
137 {
138 if(bytePosition_ >= byteCapacity_)
139 {
140 grow();
141 }
142 if(present)
143 {
144 bits_[bytePosition_] |= bitMask_;
145 }
146 else
147 {
148 bits_[bytePosition_] &= ~bitMask_;
149 }
150 if(vout_)
151 {
152 verboseSetNext(present);
153 }
154 bitMask_ >>= 1;
155 if(bitMask_ == 0)
156 {
157 bitMask_ = startByteMask;
158 bytePosition_ += 1;
159 }
160 }

162 inline
163 bool
164 PresenceMap::checkNextField()
165 {
166 if(bytePosition_ >= byteCapacity_)
167 {
168 if(vout_)(*vout_) << "pmap:at end [" << bytePosition_ << "]" << std::endl;
169 return false;
170 }
171 bool result = (bits_[bytePosition_] & bitMask_) != 0;
172 if(vout_)
173 {
174 verboseCheckNextField(result);
175 }
176 bitMask_ >>= 1;
177 if(bitMask_ == 0)
178 {
179 bitMask_ = startByteMask;
180 bytePosition_ += 1;
181 }
182 return result;
183 }

185 inline
186 bool
187 PresenceMap::checkSpecificField(size_t bit)
188 {
189 size_t byte = bit / 7;
190 if(byte >= byteCapacity_)
191 {
192 return false;
193 }
194 size_t bitNum = bit % 7;
195 unsigned char bitmask = startByteMask >> bitNum;
196 bool result = ((bits_[byte] & bitmask) != 0);
197 if(vout_)
198 {
199 verboseCheckSpecificField(bit, byte, bitmask, result);
200 }
201 return result;
202 }
