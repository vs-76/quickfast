129 bool start(
130 Assembler & assembler,
131 size_t bufferSize = 1500,
132 size_t bufferCount = 2)
133 {
134 bool result = false;
135 assembler_ = & assembler;
136 bufferSize_ = bufferSize;
137 if(initializeReceiver())
138 {
139 assembler_->receiverStarted(*this);
140
141 // Allocate initial set of buffers
142 std::unique_lock<std::mutex> lock(bufferMutex_);
143
144 for(size_t nBuffer = 0; nBuffer < bufferCount; ++nBuffer)
145 {
146 BufferLifetime buffer(new LinkedBuffer(bufferSize));
148 bufferLifetimes_.push_back(buffer);
149 idleBufferPool_.push(buffer.get());
150 }
151 startReceive(lock);
152 result = true;
153 }
154 return result;
155 }

311 virtual LinkedBuffer * getBuffer(bool wait)
312 {
313 LinkedBuffer *next = queue_.serviceNext();
314 bool more = true;
315 while(more && next == 0 && !stopping_)
316 {
317 more = wait;
318 {
319 std::unique_lock<std::mutex> lock(bufferMutex_);
320 // add any idle buffers to pool
321 idleBufferPool_.push(idleBuffers_);
322 startReceive(lock);
323 queue_.refresh(lock, wait && !stopping_ && !inputComplete_);
324 }
325 next = queue_.serviceNext();
326 }
327 if(next != 0)
328 {
329 ++packetsProcessed_;
330 bytesProcessed_ += next->used();
331 }
332 return next;
333 }
334
342 bool needBytes(size_t needed, bool wait)
343 {
344 size_t available = 0;
345 bool more = true;
346 while(available < needed && more)
347 {
348 const LinkedBuffer *next = queue_.peekOutgoing();
349 while(available < needed && next != 0)
350 {
351 available += next->used();
352 next = next->link();
353 }
354 if(available < needed && wait)
355 {
356 std::unique_lock<std::mutex> lock(bufferMutex_);
357 // add any idle buffers to pool
358 idleBufferPool_.push(idleBuffers_);
359 startReceive(lock);
360 // wait is already true in this branch (guard above).
361 queue_.refresh(lock, !inputComplete_);
362 available = 0;
363 }
364 else
365 {
366 more = false;
367 }
368 }
369 return available >= needed;
370 }

403 void startReceive(std::unique_lock<std::mutex>& lock)
404 {
405 bool more = canStartRead();
406 while( more && !stopping_)
407 {
408 more = false;
409 LinkedBuffer *buffer = idleBufferPool_.pop();
410 if(buffer != 0)
411 {
412 ++readsInProgress_;
413 if(fillBuffer(buffer, lock))
414 {
415 more = canStartRead();
416 }
417 else
418 {
419 idleBufferPool_.push(buffer);
420 --readsInProgress_;
421 inputComplete_ = true;
422 endOfInput();
423 }
424 }
425 else
426 {
427 //std::ostringstream msg;
428 //msg << "{" << (void *)this << "} Trying to read. No buffer available\n";
429 //std::cout << msg.str();
430 ++noBufferAvailable_;
431 }
432 }

617 // stop(), pause() and resume() are public, take no lock, and are called
618 // from whatever thread drives shutdown; the flags are read on the
619 // service thread and, for stopping_, on asio completion threads. As
620 // plain bools that is a data race, and the familiar consequence is that
621 // nothing tells the compiler the value can change under a spin: the
622 // load may be hoisted out of the loop and stop() never observed, so the
623 // receiver does not shut down and the multicast socket is never closed.
