33 inline
34 bool CASPtr(void * volatile * target, void * ifeq, void * value)
35 {
36#if defined(_WIN32)
37# if defined(_M_IX86) && (_M_IX86 >= 400) && ! defined(_M_IA64)
41 //#pragma message("Assembly language CASPtr")
42 __asm
43 {
44 mov eax, ifeq
45 mov ecx, value
46 mov edx, target
47 lock cmpxchg [edx], ecx
48 sete al
49 xor ah, ah
50 }
51# else
52 // non-x86/32 WIN32 platforms.
53 // note the WIN32 function takes its arguments in an unusual order.
54 // #pragma mesasge("function call CASPtr")
55 return ifeq == InterlockedCompareExchangePointer(
56 (PVOID volatile *)target, value, ifeq);
57# endif
58#elif defined(__GNUC__)
59# if defined(FORCE_ASM_ON_GCC) // force ASM on gcc (assume x86 processor)
60 bool result;
61 asm(
62 "lock \n\t"
63 "cmpxchgl %%ecx,(%%edx) \n\t"
64 "setz %%al \n\t"
65 "xor %%ah,%%ah"
66 : "=a" (result)
67 : "a" (ifeq) , "c" (value), "d"(target) /* */
68 : "memory"
69 );
70 return result;
71# else // force ASM on gcc
72 return __sync_bool_compare_and_swap(target, ifeq, value);
73# endif // force asm on gcc
74#else // otherwise we hope this is defind on your favorite platform
75 return ifeq == atomic_cas_ptr(target, ifeq, value);
76#endif
77 }
